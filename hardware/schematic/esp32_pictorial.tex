% ============================================================
% esp32_pictorial.tex — spatial / colour-coded build view.
%
% Complementary to esp32_supervisor.svg, not a substitute. This
% sheet answers "which coloured wire goes where on the bench",
% laid out the way a breadboard actually sits: the shared rails
% along the bottom, modules above them.
%
% Pin-level truth lives in WIRING.md. Signal topology lives in
% esp32_supervisor.svg. This drawing is spatial only.
%
% Note there is no +5 V rail. The board is fed through its own
% USB-C connector and nothing else in the build runs off 5 V, so
% the only shared buses are 3V3 (regulated on the DevKitC) and
% GND. The fob is on the far side of the optocoupler and shares
% neither.
%
% Routing rule: vertical wires pass OVER horizontal ones (white
% under-stroke). A filled dot is the only thing that ever means
% "connected". Every drop to a rail is placed clear of a module
% footprint — nothing routes through a block.
%
% Regenerate with:  make esp32_pictorial.svg
% ============================================================

\node[legendbox, anchor=north west, text width=8.4cm] at (-5.2,-3.6) {%
  \textbf{Reading this sheet}\\[4pt]
  A filled dot is the \emph{only} thing that means connected. Where a
  vertical wire crosses a horizontal one it breaks over the top and
  nothing is joined --- the pulldown and all three GND drops cross the
  3V3 rail this way.\\[6pt]
  \textbf{Traps worth re-reading before power-up}\\[4pt]
  \begin{tabular}{@{}p{0.35cm}@{}p{7.5cm}@{}}
    1 & The NTC must stay on GPIO34 (ADC1). ADC2 pins read zero
        while Wi-Fi is up.\\[3pt]
    2 & The 10\,k$\Omega$ pulldown on GPIO26 is not optional ---
        floating must mean not transmitting.\\[3pt]
    3 & Nothing on the fob side of the optocoupler touches the GND
        rail. Bonding them defeats the isolation.\\[3pt]
    4 & The fob pad sits on 12\,V. A direct GPIO connection destroys
        the ESP32.\\[3pt]
    5 & Measure the divider resistor and put the measured value in
        the firmware's $\beta$ equation.\\
  \end{tabular}
};
