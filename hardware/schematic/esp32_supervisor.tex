% esp32_supervisor.tex — low-voltage side of the supervisor.
%
% Drawn for the parts actually purchased, and nothing else:
%
%   B0GF1ZJCCN  ESP32-DevKitC-32E, 2-pack, USB-C, 38-pin
%   B0F8NQ9S4R  DROK 10K/B3950 NTC probe, 3-pack, JST XH2.54
%   B07CTL3TG6  VONVOFF 433 MHz RF switch kit (2 fobs + receiver)
%
% Things a reader coming from ARCHITECTURE.md's end-state design
% will not find here, because the parts do not exist yet:
%   * Power arrives through the board's USB-C connector, not VIN.
%     No isolated PSU module (TASK-2).
%   * No MAX31855 and no thermocouple (TASK-1). The DROK NTC on
%     GPIO34 is the only sensor, and here it IS the trip source.
%   * GPIO26 does not drive a relay module (TASK-3). It gates the
%     fob's ON-button pad through a level shifter, and the
%     resulting 433 MHz carrier is the heartbeat.
%   * No ack button (see BUILD-TONIGHT.md § 5 — MANUAL_LOCKOUT
%     clears by power-cycle).
%
% Layout bands:
%   y = 5.0 .. 10.4   USB supply, NTC divider, DROK probe
%   y = 1.0 ..  5.0   GPIO26 -> pulldown -> level shifter -> fob
%   y < -0.8          legends and title block
%

% ============================================================

\node[legendbox, anchor=north west, text width=8.2cm] at (-13.6,-0.8) {%
  \textbf{The inversion that makes this fail-safe}\\[4pt]
  There is no relay the ESP32 can \emph{open}. The receiver is
  programmed to \textbf{momentary} mode instead, so its contact is
  closed only while a carrier is arriving. GPIO26 HIGH $=$ fob
  transmitting $=$ contact closed $=$ the coil may be
  energised.\\[5pt]
  \begin{tabular}{@{}p{4.0cm}@{\hspace{4pt}}p{3.5cm}@{}}
    \textbf{Fault} & \textbf{Result}\\[2pt]
    ESP32 power loss        & TX stops, contact opens\\[1pt]
    Firmware hang           & watchdog stops TX\\[1pt]
    NTC open or shorted     & firmware stops TX\\[1pt]
    Fob cell dies           & TX stops\\[1pt]
    Jammed or out of range  & no carrier decoded\\[1pt]
    GPIO26 floating at boot & 10\,k$\Omega$ pulldown wins\\
  \end{tabular}\\[5pt]
  Every one of them stops the saw. Nuisance stops are the accepted
  cost of having protection at all.
};

\node[legendbox, anchor=north west, text width=8.0cm] at (-4.8,-0.8) {%
  \textbf{If no optocoupler turns up}\\[4pt]
  Same node, different part. All three keep the 10\,k$\Omega$
  pulldown on GPIO26.\\[4pt]
  \begin{tabular}{@{}p{1.75cm}@{\hspace{4pt}}p{5.45cm}@{}}
    NPN & 2N3904 / 2N2222 / BC547, 1\,k$\Omega$ base resistor,
          collector to the ON pad, emitter to fob BAT$-$.
          \textbf{Ties ESP32 GND to fob GND.}\\[3pt]
    MOSFET & 2N7000 / BSS138 / AO3400 low-side on the fob's battery
          negative, 100\,k$\Omega$ gate to source. Switches fob
          \emph{power} rather than the button.\\[3pt]
    Reed relay & Coil driven by the NPN circuit with a flyback diode,
          contacts in the fob's battery positive lead.\\
  \end{tabular}\\[4pt]
  \textcolor{tsmains}{\textbf{Never}} wire GPIO26 straight to the ON
  pad. The pad sits on the fob's 12\,V rail and will destroy the
  ESP32.
};

\node[legendbox, anchor=north west, text width=7.4cm] at (4.0,-0.8) {%
  \textbf{Bench checks before this goes near the saw}\\[4pt]
  \begin{tabular}{@{}p{0.35cm}@{}p{6.5cm}@{}}
    1 & Measure the 10\,k$\Omega$ divider resistor and put the
        \emph{actual} value into the $\beta$ equation. A 5\,\% part
        can measure 9.5--10.5\,k$\Omega$.\\[3pt]
    2 & Two-point calibrate in ice water and boiling water before
        the probe is mounted (§ 5).\\[3pt]
    3 & With the ESP32 unpowered, confirm the fob is \emph{not}
        transmitting and the receiver contact is open.\\[3pt]
    4 & GPIO34 is input-only and on ADC1. Do not move the divider
        to an ADC2 pin --- those read zero once Wi-Fi is up.\\[3pt]
    5 & Watch the contact for 60\,s with the probe cold. Any
        drop-out means the encoder is not producing a continuous
        frame stream.\\
  \end{tabular}
};
